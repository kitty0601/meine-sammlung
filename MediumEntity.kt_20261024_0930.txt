
package com.example.meinesammlung

import androidx.room.Entity
import androidx.room.PrimaryKey

@Entity(tableName = "media")
data class MediumEntity(
    @PrimaryKey(autoGenerate = true)
    val id: Long = 0,

    val title: String,
    val originalTitle: String = "",
    val mediumType: String = "DVD",
    val genre: String = "",
    val releaseDateGermany: String = "",
    val people: String = "",
    val barcode: String = "",
    val edition: String = "",
    val notes: String = "",
    val coverUrl: String = ""
)
